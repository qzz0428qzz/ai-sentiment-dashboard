import pandas as pd
import numpy as np
from pathlib import Path
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, LabelEncoder

DATA_FILE = Path('/workspace/.uploads/23091186-5ec3-4346-9498-dc542bc22971_student_data_messy.csv')

df = pd.read_csv(DATA_FILE)
clean = df.copy()

# 列名规范化
clean.columns = clean.columns.str.strip().str.lower().str.replace(' ', '_', regex=False)

# 缺失值符号统一
clean = clean.replace(['', ' ', 'NA', 'N/A', 'na', 'n/a', '?', 'missing', 'Missing'], np.nan)

# 文本去首尾空格
for col in clean.select_dtypes(include=['object', 'string']).columns:
    clean[col] = clean[col].astype('string').str.strip()

# 统一性别
clean['gender'] = clean['gender'].str.lower().map({'m': 'Male', 'male': 'Male', 'f': 'Female', 'female': 'Female'})

# 统一专业
major_key = clean['major'].str.lower().str.replace('.', '', regex=False)
clean['major'] = major_key.map({
    'cs': 'Computer Science', 'computer science': 'Computer Science',
    'ai': 'Artificial Intelligence', 'artificial intelligence': 'Artificial Intelligence',
    'data science': 'Data Science',
})

# 统一城市大小写
clean['city'] = clean['city'].str.title()

# 数值列转换
numeric_cols = ['age', 'attendance_pct', 'homework', 'presentation', 'project', 'height_cm', 'weight_kg']
for col in numeric_cols:
    clean[col] = pd.to_numeric(clean[col], errors='coerce')

# 数值范围校验，超出置为缺失
ranges = {
    'age': (16, 80), 'attendance_pct': (0, 100), 'homework': (0, 20),
    'presentation': (0, 20), 'project': (0, 50), 'height_cm': (130, 220), 'weight_kg': (35, 200),
}
for col, (low, high) in ranges.items():
    clean.loc[~clean[col].between(low, high), col] = np.nan

# 日期解析
clean['enrollment_date'] = pd.to_datetime(clean['enrollment_date'], errors='coerce', format='mixed')

# 删除重复行
clean = clean.drop_duplicates().reset_index(drop=True)

# 缺失值填充（数值用中位数，类别用众数）
clean[numeric_cols] = SimpleImputer(strategy='median').fit_transform(clean[numeric_cols])
for col in ['gender', 'major', 'city']:
    clean[col] = clean[col].fillna(clean[col].mode(dropna=True)[0])

# 邮箱清理与有效性
clean['email'] = clean['email'].str.strip().str.lower()
clean['email_valid'] = clean['email'].str.contains('@', na=False)

# 类型调整
clean['age'] = clean['age'].round().astype('Int64')
for col in ['gender', 'major', 'city']:
    clean[col] = clean[col].astype('category')

# 数值标准化
scale_cols = ['attendance_pct', 'homework', 'presentation', 'project', 'height_cm', 'weight_kg']
scaled = pd.DataFrame(
    StandardScaler().fit_transform(clean[scale_cols]),
    columns=[f'{c}_scaled' for c in scale_cols], index=clean.index,
)
clean = pd.concat([clean, scaled], axis=1)

# 独热编码（city）
clean = pd.get_dummies(clean, columns=['city'], prefix='city', dtype=int)

# 标签编码（gender, major）
clean['gender_encoded'] = LabelEncoder().fit_transform(clean['gender'])
clean['major_encoded'] = LabelEncoder().fit_transform(clean['major'])

# 保存结果
clean.to_csv('/workspace/student_data_cleaned.csv', index=False)
clean.to_excel('/workspace/student_data_cleaned.xlsx', index=False)
